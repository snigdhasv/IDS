QUICK START (Copy & Paste)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 1. Configure DPDK
bash dpdk_suricata_ml_pipeline/scripts/04_configure_dpdk_input.sh

# 2. Start Pipeline
sudo bash run_realtime_engine_dpdk.sh start

# 3. Replay PCAP
python3 dpdk_pcap_replay.py \
  dpdk_suricata_ml_pipeline/pcap_samples/mixed_traffic_sample.pcap \
  --csv results/ground_truth.csv

# 4. Monitor (in another terminal)
tail -f logs/ml_consumer.log

# 5. Calculate Accuracy
python3 calculate_accuracy_metrics.py \
  --packets results/ground_truth.csv \
  --predictions logs/ml_consumer.log \
  --output results/accuracy.json

# 6. View Results
cat results/accuracy.json | jq .overall.accuracy

# 7. Stop Pipeline
sudo bash run_realtime_engine_dpdk.sh stop

Scenario 1 - Quick Test (5 min):
  sudo bash run_realtime_engine_dpdk.sh start
  python3 dpdk_pcap_replay.py normal_traffic.pcap --csv gt.csv
  python3 calculate_accuracy_metrics.py --packets gt.csv --predictions logs/ml_consumer.log

Scenario 2 - Full Test (30 min):
  sudo bash test_dpdk_replay.sh start
  python3 calculate_accuracy_metrics.py --packets test_results/*.csv --predictions test_results/ml_predictions.log

Scenario 3 - Stress Test (1+ hour):
  sudo bash run_realtime_engine_dpdk.sh start
  python3 dpdk_pcap_replay.py mixed_traffic_sample.pcap --repeat 100 --rate 1000000 --csv stress.csv
  python3 calculate_accuracy_metrics.py --packets stress.csv --predictions logs/ml_consumer.log

Scenario 4 - Real-time Monitoring:
  Terminal 1: sudo bash run_realtime_engine_dpdk.sh start
  Terminal 2: tail -f logs/ml_consumer.log
  Terminal 3: python3 dpdk_pcap_replay.py mixed_traffic_sample.pcap
